PROBLEM:
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
Vercel can't connect to localhost:5000 because:
- localhost = YOUR computer only
- Vercel servers are on the internet
- They can't reach YOUR localhost

SOLUTION:
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
Expose your backend using ngrok

STEP 1: Download ngrok
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
https://ngrok.com/download
Extract ngrok.exe

STEP 2: Run ngrok
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
Double-click: start-ngrok.bat

OR manually:
ngrok http 5000

STEP 3: Copy the HTTPS URL
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
You'll see:
Forwarding  https://abc123.ngrok-free.app -> http://localhost:5000

📋 COPY: https://abc123.ngrok-free.app

STEP 4: Update Vercel Environment Variable
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
Go to: https://vercel.com
→ Your Project
→ Settings
→ Environment Variables

Add/Update:
Name: NEXT_PUBLIC_API_URL
Value: https://abc123.ngrok-free.app (YOUR URL)
Environment: ✓ Production

Click "Save"

STEP 5: Update Backend CORS
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
Edit: server/src/app.js (line 85)

Add your ngrok URL:
const allowedOrigins = [
  'http://localhost:3000',
  process.env.CLIENT_URL || 'http://localhost:3000',
  'https://smart-lms-clean.vercel.app',
  'https://abc123.ngrok-free.app', // ADD THIS
  ''
];

Restart backend:
cd server
Ctrl+C
npm start

STEP 6: Redeploy Vercel
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
cd client
vercel --prod

OR push to GitHub (auto-deploys)
